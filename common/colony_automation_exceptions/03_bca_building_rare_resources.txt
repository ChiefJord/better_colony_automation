automate_crystal_deficit_management = {
	available = {
		bca_has_main_resource_to_build = yes
		owner = {
			OR = {
				resource_income_compare = { resource = rare_crystals value < 0 }
				AND = {
					has_country_flag = bca_enable_economic_management
					bca_country_needs_rare_crystals = yes
				}
			}
		}
		OR = {
			has_any_industrial_designation = yes
			AND = {
				free_building_slots > 0
				has_research_designation = no
				has_unity_designation = no
				has_fortress_designation = no
			}
		}
	}
	emergency = yes
	category = "planet_automation_rare_resources"

	buildings = {
		1 = {
			building = building_archaeo_refinery
		}
		2 = {
			building = building_crystal_plant
			available = {
				OR = {
					owner = { has_country_flag = bca_retain_low_level_rare_resource_buildings }
					NOT = { has_building = building_archaeo_refinery }
				}
			}
		}
	}
}
